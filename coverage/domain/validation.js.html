const toInteger = (value) => {
  if (typeof value === 'number') return Number.isInteger(value) ? value : NaN
  const text = typeof value === 'string' ? value.trim() : ''
  return /^-?\d+$/.test(text) ? Number(text) : NaN
}
 
export const validateQty = (value) => {

}
